cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
project(WcfiControlNode VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WCFI Control Node prototype requires Windows x64")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory(../wcfi_analysis_node/deps/vst3sdk sdk)
smtg_enable_vst3_sdk()
smtg_add_vst3plugin(WcfiControlNode PACKAGE_NAME "WCFI Control Node RED"
    SOURCES_LIST node.cpp node.h dsp.cpp dsp.h single_mic.cpp single_mic.h editor.cpp editor.h factory.cpp)
target_compile_features(WcfiControlNode PRIVATE cxx_std_20)
target_link_libraries(WcfiControlNode PRIVATE sdk user32 gdi32)
target_compile_definitions(WcfiControlNode PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WcfiControlNode PRIVATE /Z7 /fp:precise /utf-8)
add_library(WcfiControlBridge SHARED dsp.cpp dsp.h)
target_compile_features(WcfiControlBridge PRIVATE cxx_std_20)
target_compile_definitions(WcfiControlBridge PRIVATE WCFI_CONTROL_EXPORTS)
target_compile_options(WcfiControlBridge PRIVATE /Z7 /fp:precise)
set_target_properties(WcfiControlBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(WcfiMicBridge SHARED single_mic.cpp single_mic.h dsp.cpp dsp.h)
target_compile_features(WcfiMicBridge PRIVATE cxx_std_20)
target_compile_definitions(WcfiMicBridge PRIVATE WCFI_CONTROL_EXPORTS WCFI_MIC_EXPORTS)
target_compile_options(WcfiMicBridge PRIVATE /Z7 /fp:precise)
set_target_properties(WcfiMicBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(control_tests tests.cpp node.cpp node.h dsp.cpp dsp.h single_mic.cpp single_mic.h editor.cpp editor.h
    ../wcfi_analysis_node/deps/vst3sdk/public.sdk/source/common/memorystream.cpp)
target_compile_features(control_tests PRIVATE cxx_std_20)
target_link_libraries(control_tests PRIVATE sdk sdk_hosting user32 gdi32)
target_compile_definitions(control_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(control_tests PRIVATE /Z7 /fp:precise /utf-8)
add_test(NAME control_node COMMAND control_tests)
add_executable(single_mic_tests single_mic_tests.cpp single_mic.cpp single_mic.h dsp.cpp dsp.h)
target_compile_features(single_mic_tests PRIVATE cxx_std_20)
target_compile_options(single_mic_tests PRIVATE /Z7 /fp:precise /utf-8)
add_test(NAME single_mic COMMAND single_mic_tests)

# Standalone view of the actual editor, with a fake host and no audio I/O.
add_executable(control_ui_preview WIN32 preview.cpp node.cpp node.h dsp.cpp dsp.h single_mic.cpp single_mic.h editor.cpp editor.h)
target_compile_features(control_ui_preview PRIVATE cxx_std_20)
target_link_libraries(control_ui_preview PRIVATE sdk sdk_hosting user32 gdi32)
target_compile_definitions(control_ui_preview PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(control_ui_preview PRIVATE /Z7 /fp:precise /utf-8)
set_target_properties(control_ui_preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(control_ui_tests ui_tests.cpp node.cpp node.h dsp.cpp dsp.h single_mic.cpp single_mic.h editor.cpp editor.h)
target_compile_features(control_ui_tests PRIVATE cxx_std_20)
target_link_libraries(control_ui_tests PRIVATE sdk sdk_hosting user32 gdi32)
target_compile_definitions(control_ui_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(control_ui_tests PRIVATE /Z7 /fp:precise /utf-8)
add_test(NAME control_node_ui COMMAND control_ui_tests)
