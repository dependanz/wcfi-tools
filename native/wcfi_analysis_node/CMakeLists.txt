cmake_minimum_required(VERSION 3.25)
project(WcfiAnalysisNode VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The initial WCFI Analysis Node prototype requires Windows x64")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory(deps/vst3sdk)
smtg_enable_vst3_sdk()
smtg_add_vst3plugin(WcfiAnalysisNode PACKAGE_NAME "WCFI Analysis Node"
    SOURCES_LIST node.cpp node.h editor.cpp factory.cpp transport.cpp transport.h)
target_compile_features(WcfiAnalysisNode PRIVATE cxx_std_20)
target_link_libraries(WcfiAnalysisNode PRIVATE sdk ole32 advapi32 user32 gdi32)
target_compile_definitions(WcfiAnalysisNode PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(WcfiAnalysisBridge SHARED transport.cpp transport.h)
target_compile_features(WcfiAnalysisBridge PRIVATE cxx_std_20)
target_link_libraries(WcfiAnalysisBridge PRIVATE ole32 advapi32)
enable_testing()
add_executable(node_tests tests.cpp node.cpp editor.cpp transport.cpp
    deps/vst3sdk/public.sdk/source/common/memorystream.cpp)
target_compile_features(node_tests PRIVATE cxx_std_20)
target_link_libraries(node_tests PRIVATE sdk sdk_hosting ole32 advapi32 user32 gdi32)
target_compile_definitions(node_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME analysis_node COMMAND node_tests)
